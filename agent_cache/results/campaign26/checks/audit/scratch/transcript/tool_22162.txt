const results = await pipeline(
  LENSES,
  lens => agent(`${lens.prompt}\n${SHARED}\nReturn your assessment through the structured output: an overall judgement from your lens, the strengths you verified, and the findings (most severe first; at most 12; only real problems, each with evidence you checked yourself and a concrete fix).`,
    { label: `review:${lens.key}`, phase: 'Review', schema: FINDINGS }),
  (review, lens) => {
    if (!review) return null
    return agent(`You are a skeptic checking another reviewer's work. The reviewer had this lens:\n\n${lens.prompt}\n\nTheir findings, as JSON:\n\n${JSON.stringify(review, null, 1)}\n\nYOUR JOB: try to refute each finding. Check its evidence yourself in the cited files and slides, redo any derivation or number, and look for counter-evidence, including places where the deck or the plan already handles the point. Verdict "confirmed" only if you reproduced the evidence; "partly" if the core holds but the statement or its severity needs correcting (give the corrected statement); "refuted" if the evidence does not hold or the point is already handled (say where). Default to "refuted" when you cannot reproduce the evidence. Then judge the proposed fix: would it work, and is it the smallest change? Correct it if not. Finally list up to three important problems within this lens that the reviewer missed, each with evidence and a fix.\n${SHARED}\nUse the lens name "${lens.key}_verify" for your scratch folder.`,
      { label: `verify:${lens.key}`, phase: 'Verify', schema: VERDICTS })
      .then(verification => ({ lens: lens.key, review, verification }))
  }
)

return results.filter(Boolean)
