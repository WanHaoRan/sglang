const verifyPrompt = (key, i, f) => RULES(V + '/verify_' + key + '_' + (i + 1)) + DATA + `

const results = await pipeline(
  TASKS,
  t => agent(t.prompt, { label: t.kind + ':' + t.key, phase: t.phase, schema: t.schema }),
  (res, t) => {
    if (!res) return { key: t.key, result: null }
    if (t.kind !== 'attack') return { key: t.key, result: res }
    const serious = (res.findings || []).filter(f => f.severity !== 'low' && f.checkable_from_logs)
    const picked = serious.slice(0, t.cap)
    if (serious.length > picked.length) log(t.key + ': ' + (serious.length - picked.length) + ' serious findings not sent to a verifier (cap ' + t.cap + ')')
    return parallel(picked.map((f, i) => () =>
      agent(verifyPrompt(t.key, i, f), { label: 'verify:' + t.key + ':' + (i + 1), phase: 'Verify', schema: VERDICT_SCHEMA })
        .then(v => ({ finding: f.title, verdict: v }))))
      .then(vs => ({ key: t.key, result: res, verified: vs.filter(Boolean) }))
  })
return results
